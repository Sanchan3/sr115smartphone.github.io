let mediaRecorder; let recordedChunks = []; // 共通設定（初期化時に一度だけ実行） async function setupAudio() { const audioContext = new (window.AudioContext || window.webkitAudioContext)(); await audioContext.audioWorklet.addModule('processor.js'); const stream = await navigator.mediaDevices.getUserMedia({ audio: true }); const microphone = audioContext.createMediaStreamSource(stream); const robotNode = new AudioWorkletNode(audioContext, 'pass-through-processor'); // 録音データを貯めるための専用ノード const streamDestination = audioContext.createMediaStreamDestination(); return { audioContext, microphone, robotNode, streamDestination }; } // モードを開始する関数 function startVoiceAction(mode, audioSetup) { const { audioContext, microphone, robotNode, streamDestination } = audioSetup; // まずマイクとロボットボイスを繋ぐ microphone.connect(robotNode); // 接続を一旦リセット robotNode.disconnect(); if (mode === 'realtime') { // 【リアルタイムモード】直接スピーカーに出力 robotNode.connect(audioContext.destination); } else if (mode === 'record') { // 【録音モード】スピーカーには繋がず、録音用ノードに出力 robotNode.connect(streamDestination); recordedChunks = []; mediaRecorder = new MediaRecorder(streamDestination.stream); mediaRecorder.ondataavailable = (event) => { if (event.data.size > 0) recordedChunks.push(event.data); }; mediaRecorder.start(); } } // 録音モードの時に「話し終わった」タイミングで呼ぶ関数 function stopAndPlayRecord() { if (mediaRecorder && mediaRecorder.state === 'recording') { mediaRecorder.onstop = () => { // 録音したロボットボイスを1つの音声ファイルに結合して再生 const blob = new Blob(recordedChunks, { type: 'audio/webm' }); const audioUrl = URL.createObjectURL(blob); const audio = new Audio(audioUrl); audio.play(); // ここで話し終わった音声が一気に再生される }; mediaRecorder.stop(); } }
